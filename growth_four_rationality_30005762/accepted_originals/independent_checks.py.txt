"""Independent regression checks, not a proof of any infinite statement."""
from itertools import permutations, combinations
import sympy as s
import json
from pathlib import Path
x=s.symbols('x')
q=[s.Integer(1),s.Integer(1)]
for k in range(2,51):
 q.append(s.expand(q[-1]-x*q[-2]))
for k in range(1,51):
 closed=sum((-x)**j*s.binomial(k-j,j) for j in range(k//2+1))
 assert s.expand(closed-q[k])==0
 assert s.degree(q[k],x)==k//2
 assert s.gcd(q[k],q[k-1])==1
counts={k:[] for k in range(2,9)}
for n in range(9):
 c={k:0 for k in counts}
 for p in permutations(range(n)):
  if any(p[i]<p[l]<p[j] for i,j,l in combinations(range(n),3)):
   continue
  lis=[]
  for i,a in enumerate(p):
   lis.append(1+max((lis[j] for j in range(i) if p[j]<a),default=0))
  m=max(lis,default=0)
  for k in c:
   c[k]+=m<k
 for k in c: counts[k].append(c[k])
for k,cs in counts.items():
 series=s.series(q[k-1]/q[k],x,0,9).removeO().expand()
 assert cs==[int(series.coeff(x,n)) for n in range(9)]
result={'status':'PASS','symbolic_k_range':[1,50],'direct_permutation_n_range':[0,8],'direct_permutation_k_range':[2,8],'counts':counts,'warning':'Finite regression checks only. Infinite conclusions depend on the combinatorial and algebraic proof.'}
Path(__file__).with_name('independent_checks.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result))
